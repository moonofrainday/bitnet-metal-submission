cmake_minimum_required(VERSION 3.20)
project(bitnet_actquant_validation LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(BUILD_SHARED_LIBS ON CACHE BOOL "" FORCE)
set(GGML_METAL ON CACHE BOOL "" FORCE)
set(GGML_METAL_EMBED_LIBRARY ON CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS ON CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON ON CACHE BOOL "" FORCE)
set(LLAMA_BUILD_APP OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_MTMD OFF CACHE BOOL "" FORCE)
set(LLAMA_OPENSSL OFF CACHE BOOL "" FORCE)
get_filename_component(BITNET_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
add_subdirectory("${BITNET_ROOT}" upstream)
add_executable(bitnet-actquant-runner runner.cpp)
target_include_directories(bitnet-actquant-runner PRIVATE "${BITNET_ROOT}/vendor")
target_compile_options(bitnet-actquant-runner PRIVATE -Wall -Wextra -Werror -Wno-deprecated-declarations -fno-fast-math -ffp-contract=off)
target_link_libraries(bitnet-actquant-runner PRIVATE llama ggml ggml-cpu ggml-metal)
set_target_properties(bitnet-actquant-runner PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
